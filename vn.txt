#!/usr/bin/env python3
"""
OPERAZIONE GHOST PROTOCOL - Interceptor Module
Classificazione: EYES ONLY
Descrizione: Sentinella armata contro redirect e tracking
"""

import asyncio

class GhostInterceptor:
    """Sentinella Ghost - Blocca ogni movimento nemico"""
    
    def __init__(self):
        # ARSENALE ANTI-TRACKING
        self.blocked_redirects = 0
        self.blocked_js_redirects = 0
        self.blocked_meta_refresh = 0
        self.threat_intelligence_db = set()
        self.last_ti_update = None
        
        # PATTERN DI ATTACCO NEMICO (da MITRE ATT&CK)
        self.redirect_patterns = [
            r'window\.location\.href\s*=',
            r'window\.location\.replace\s*\(',
            r'document\.location\s*=',
            r'location\.assign\s*\(',
            r'history\.replaceState',
            r'<meta\s+http-equiv=["\']refresh["\']',
            r'setTimeout.*location',
            r'setInterval.*location'
        ]
        
        print("🛡️  GHOST INTERCEPTOR ATTIVATO - Modalità Invisibile")
        self.update_threat_intelligence()
    
    def update_threat_intelligence(self):
        """Aggiornamento intelligence da fonti militari"""
        try:
            print("📡 Aggiornamento Threat Intelligence...")
            
            # FONTE: Abuse.ch (come CIA)
            malware_urls = requests.get(
                "https://urlhaus-api.abuse.ch/v1/urls/recent/",
                timeout=10
            ).json()
            
            for entry in malware_urls.get('urls', []):
                if entry.get('url'):
                    self.threat_intelligence_db.add(entry['url'])
            
            # FONTE: MITRE ATT&CK (tattiche nemiche)
            mitre_indicators = [
                "bit.ly", "tinyurl.com", "t.co", "goo.gl",  # URL shorteners
                "doubleclick.net", "googleadservices.com",  # Tracking
                "facebook.com/tr", "google-analytics.com"   # Analytics
            ]
            
            self.threat_intelligence_db.update(mitre_indicators)
            self.last_ti_update = datetime.now()
            
            print(f"✅ Database aggiornato: {len(self.threat_intelligence_db)} indicatori")
            
        except Exception as e:
            print(f"⚠️  Errore intelligence: {e}")
    
    def is_threat_url(self, url):
        """Analisi threat intelligence (come NSA)"""
        for indicator in self.threat_intelligence_db:
            if indicator in url:
                return True
        return False
    
    def request(self, flow: http.HTTPFlow) -> None:
        """Checkpoint di entrata - Ispezione pacchetti"""
        
        # CONTROLLO BLACKLIST
        if self.is_threat_url(flow.request.pretty_url):
            print(f"🚫 BLOCCATO - Threat Intelligence: {flow.request.pretty_url}")
            flow.response = http.Response.make(
                204,  # No Content
                b"BLOCKED BY GHOST PROTOCOL",
                {"X-Ghost-Block": "threat-intelligence"}
            )
            return
        
        # ANTI-FINGERPRINTING (tecnica TEMPEST)
        ghost_headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
            "Accept-Language": "en-US,en;q=0.9",
            "Accept-Encoding": "gzip, deflate, br",
            "DNT": "1",
            "Sec-Fetch-Site": "none",
            "Sec-Fetch-Mode": "navigate"
        }
        
        # SOSTITUZIONE HEADERS (modalità fantasma)
        for header, value in ghost_headers.items():
            flow.request.headers[header] = value
        
        # RIMOZIONE TRACKING HEADERS
        tracking_headers = [
            "X-Forwarded-For", "X-Real-IP", "X-Client-IP",
            "Client-IP", "X-Cluster-Client-IP"
        ]
        
        for header in tracking_headers:
            if header in flow.request.headers:
                del flow.request.headers[header]
    
    def response(self, flow: http.HTTPFlow) -> None:
        """Checkpoint di uscita - Scansione risposta nemica"""
        
        # BLOCCO REDIRECT HTTP (3XX)
        if 300 <= flow.response.status_code < 400:
            print(f"🚫 REDIRECT HTTP BLOCCATO: {flow.response.status_code}")
            self.blocked_redirects += 1
            
            flow.response = http.Response.make(
                200,
                b"REDIRECT BLOCKED BY GHOST PROTOCOL",
                {"X-Ghost-Block": "http-redirect"}
            )
            return
        
        # SCANSIONE CONTENUTO (ispezione profonda)
        if flow.response.content:
            content = flow.response.get_text()
            if content:
                modified = False
                
                # NEUTRALIZZAZIONE JAVASCRIPT REDIRECT
                for pattern in self.redirect_patterns:
                    if re.search(pattern, content, re.IGNORECASE):
                        print(f"🚫 JS REDIRECT NEUTRALIZZATO: {pattern}")
                        content = re.sub(pattern, '// BLOCKED BY GHOST', content, flags=re.IGNORECASE)
                        self.blocked_js_redirects += 1
                        modified = True
                
                # NEUTRALIZZAZIONE META REFRESH
                meta_refresh_pattern = r'<meta\s+http-equiv=["\']refresh["\'][^>]*>'
                if re.search(meta_refresh_pattern, content, re.IGNORECASE):
                    print("🚫 META REFRESH NEUTRALIZZATO")
                    content = re.sub(meta_refresh_pattern, '<!-- BLOCKED BY GHOST -->', content, flags=re.IGNORECASE)
                    self.blocked_meta_refresh += 1
                    modified = True
                
                if modified:
                    flow.response.set_text(content)
                    flow.response.headers["X-Ghost-Modified"] = "true"
    
    def get_stats(self):
        """Rapporto missione"""
        return {
            "http_redirects_blocked": self.blocked_redirects,
            "js_redirects_blocked": self.blocked_js_redirects,
            "meta_refresh_blocked": self.blocked_meta_refresh,
            "threat_db_size": len(self.threat_intelligence_db),
            "last_ti_update": str(self.last_ti_update)
        }

"""
GHOST PROTOCOL TEST SUITE
Test di invisibilità e anti-fingerprinting

import requests
import json
import time
from urllib.parse import urlparse

class GhostTester:
    """Suite di test per verificare invisibilità"""
    
    def __init__(self):
        self.session = requests.Session()
        self.session.proxies = {
            'http': 'http://127.0.0.1:8080',
            'https': 'http://127.0.0.1:8080'
        }
        
    def test_redirect_blocking(self):
        """Test blocco redirect (manovra evasiva)"""
        print("\n🧪 TEST BLOCCO REDIRECT")
        
        test_urls = [
            "http://httpbin.org/redirect/3",  # HTTP redirect
            "http://httpbin.org/absolute-redirect/2",
            "http://httpbin.org/relative-redirect/1"
        ]
        
        for url in test_urls:
            try:
                response = self.session.get(url, allow_redirects=False, timeout=10)
                if response.status_code == 200:
                    print(f"✅ REDIRECT BLOCCATO: {url}")
                elif 300 <= response.status_code < 400:
                    print(f"❌ REDIRECT NON BLOCCATO: {url}")
                else:
                    print(f"⚠️  Risposta inaspettata: {response.status_code}")
            except Exception as e:
                print(f"❌ Errore test: {e}")
    
    def test_fingerprint_resistance(self):
        """Test anti-fingerprinting (modalità fantasma)"""
        print("\n👻 TEST ANTI-FINGERPRINTING")
        
        try:
            # Test con servizio fingerprinting
            response = self.session.get("http://httpbin.org/headers", timeout=10)
            headers = response.json().get('headers', {})
            
            # Verifica headers fantasma
            user_agent = headers.get('User-Agent', '')
            if 'Windows NT' in user_agent:
                print("✅ User-Agent mascherato (Windows)")
            else:
                print(f"⚠️  User-Agent: {user_agent}")
            
            if headers.get('Dnt') == '1':
                print("✅ Do Not Track attivo")
            else:
                print("❌ Do Not Track non rilevato")
                
        except Exception as e:
            print(f"❌ Errore test fingerprint: {e}")
    
    def test_ip_anonymity(self):
        """Test anonimato IP (copertura identità)"""
        print("\n🎭 TEST ANONIMATO IP")
        
        try:
            # IP originale (senza VPN)
            original_ip = requests.get("http://httpbin.org/ip", timeout=10).json()['origin']
            
            # IP tramite Ghost Protocol
            ghost_ip = self.session.get("http://httpbin.org/ip", timeout=10).json()['origin']
            
            if original_ip != ghost_ip:
                print(f"✅ IP MASCHERATO")
                print(f"   Originale: {original_ip}")
                print(f"   Ghost:     {ghost_ip}")
            else:
                print("❌ IP NON MASCHERATO - Verificare VPN")
                
        except Exception as e:
            print(f"❌ Errore test IP: {e}")
